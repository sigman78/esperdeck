cmake_minimum_required(VERSION 3.20)
project(idfsim_contract_tests C)
set(CMAKE_C_STANDARD 11)
if(NOT WIN32)
    message(FATAL_ERROR "These adapter tests require Windows")
endif()
add_subdirectory(../../idfsim idfsim)
add_executable(test_tasks test_tasks.c)
target_link_libraries(test_tasks PRIVATE idfsim)
enable_testing()
add_test(NAME task_entry COMMAND test_tasks)
set_tests_properties(task_entry PROPERTIES TIMEOUT 15)
add_executable(test_errors test_errors.c)
target_link_libraries(test_errors PRIVATE idfsim)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME fatal_error_check COMMAND ${Python3_EXECUTABLE}
         ${CMAKE_CURRENT_SOURCE_DIR}/test_errors.py $<TARGET_FILE:test_errors>)
set_tests_properties(fatal_error_check PROPERTIES TIMEOUT 15)
